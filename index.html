SSTACK SEGMENT STACK
DW 32 DUP(?)
SSTACK ENDS
CODE SEGMENT
ASSUME CS:CODE
START: MOV AX, 00H
       MOV DX, 600H  ; 输出端口地址600H
       MOV AL, 00H   ; 三角波起始值0
       MOV BL, 01H   ; 增减方向标志：1=递增，0=递减
AA1:   OUT DX, AL    ; 输出当前数值到端口
       CALL DELAY    ; 延时控制三角波频率
       CMP BL, 01H   ; 判断当前是递增还是递减
       JE INC_VAL    ; 是递增则跳转到INC_VAL
DEC_VAL:             ; 递减分支
       DEC AL        ; 数值减1
       CMP AL, 00H   ; 判断是否到最小值0
       JE CHANGE_INC ; 到0则切换为递增
       JMP AA1
INC_VAL:             ; 递增分支
       INC AL        ; 数值加1
       CMP AL, 7FH   ; 判断是否到最大值7FH（与原代码峰值一致）
       JE CHANGE_DEC ; 到峰值则切换为递减
       JMP AA1
CHANGE_INC:
       MOV BL, 01H   ; 置为递增标志
       JMP AA1
CHANGE_DEC:
       MOV BL, 00H   ; 置为递减标志
       JMP AA1
; 延时子程序（与原代码一致，可调整CX值改变频率）
DELAY: PUSH CX
       MOV CX, 0FF00H
AA2:   PUSH AX
       POP AX
       LOOP AA2
       POP CX
       RET
CODE ENDS
END START
