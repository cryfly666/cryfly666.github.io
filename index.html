GEN_TRIANGLE:
    MOV DX, DAC0832
    MOV BL, 0       ; 当前值初始为0
    MOV BH, 1       ; 方向标志：1=递增，0=递减
TRI_LOOP:
    MOV AL, BL      ; 把当前值送入AL准备输出
    OUT DX, AL      ; 输出到DAC0832
    CALL FREQ_DELAY ; 延时控制频率
    CMP BH, 1
    JE TRI_INC      ; 递增分支
TRI_DEC:
    DEC BL          ; 数值减1
    CMP BL, 0
    JE SET_INC      ; 到0则切换为递增
    JMP TRI_LOOP
TRI_INC:
    INC BL          ; 数值加1
    CMP BL, 255
    JE SET_DEC      ; 到255则切换为递减
    JMP TRI_LOOP
SET_INC:
    MOV BH, 1
    JMP TRI_LOOP
SET_DEC:
    MOV BH, 0
    JMP TRI_LOOP
WAVE_END:
    ; 可放波形循环逻辑
