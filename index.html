DATAS SEGMENT
	SwitchVal DB 0
	XValue DB 0
	Freq DB 56H
	Power DB 01H
	FBV EQU 00000001B
    SJV EQU 00000010B
    JCV EQU 00000100B
    FQV EQU 00001000B
DATAS ENDS     


SSTACK SEGMENT STACK
	DW 32 DUP(?)
SSTACK ENDS


CODE SEGMENT
	ASSUME DS:DATAS, SS:SSTACK, CS:CODE
	
START: 
	MOV AX, DATAS
	MOV DS, AX

	MOV DX, 0646H
	MOV AL, 90H
	OUT DX, AL
	
;中断地址设置
	PUSH DS
	MOV AX, 0000H
	MOV DS, AX
	MOV AX, OFFSET MIR7
	MOV SI, 003CH
	MOV [SI], AX
	MOV AX, CS
	MOV SI, 003EH
	MOV [SI], AX
	CLI
	POP DS
	
;初始化主片8259
	MOV AL, 11H
	OUT 20H, AL
	MOV AL, 08H
	OUT 21H, AL
	MOV AL, 04H
	OUT 21H, AL
	MOV AL, 01H
	OUT 21H, AL
	MOV AL, 6FH
	OUT 21H, AL
	STI
	
;获取操作符 放入XValue变量
GetSwitchVal:
	MOV BH, Power
	TEST BH, 01H
	JZ GetSwitchVal
	MOV DX, 0640H
	IN AL, DX
	MOV SwitchVal, AL
	CALL DELAY

;X轴往复
XScan: 
	INC XValue
	TEST SwitchVal, FQV
	JZ OriginalFreq
	CMP Freq, 56H
	JNE Nochange0
	MOV BH, XValue
	SHR BH, 1
	MOV XValue, BH
NoChange0:
	MOV Freq, 2BH
	JMP AFreq
OriginalFreq:
	CMP Freq, 2BH
	JNE Nochange1
	MOV BH, XValue
	SHL BH, 1
	MOV XValue, BH
NoChange1:
	MOV Freq, 56H

AFreq:
	MOV BL, Freq
	CMP XValue, BL
	JNE Main
	MOV XValue, 0
	
Main:
	MOV AL, 0
	MOV BH, Freq
	SHR BH, 1
	CMP XValue, BH
	JA SquareWave1
	JMP SquareWave0

SquareWave0:
	TEST SwitchVal, FBV
	JZ TriangleWave0
	ADD AL, 56H
	
TriangleWave0:
	TEST SwitchVal, SJV
	JZ SawtoothWave0
	ADD AL, XValue
	ADD AL, XValue
	TEST SwitchVal, FQV
	JZ SawtoothWave0
	ADD AL, XValue
	ADD AL, XValue
	
SawtoothWave0:
	TEST SwitchVal, JCV
	JZ LoopEnd
	ADD AL, XValue
	TEST SwitchVal, FQV
	JZ LoopEnd
	ADD AL, XValue
	JMP LoopEnd
	
SquareWave1:
	TEST SwitchVal, FBV
	JZ TriangleWave1
	
TriangleWave1:
	TEST SwitchVal, SJV
	JZ SawtoothWave1
	MOV BL, Freq
	SUB BL, XValue
	ADD AL, BL
	ADD AL, BL
	TEST SwitchVal, FQV
	JZ SawtoothWave1
	ADD AL, BL
	ADD AL, BL
	
SawtoothWave1:
	TEST SwitchVal, JCV
	JZ LoopEnd
	ADD AL, XValue
	TEST SwitchVal, FQV
	JZ LoopEnd
	ADD AL, XValue

LoopEnd:
	MOV DX, 600H
	OUT DX, AL
	JMP GetSwitchVal

MIR7: 
	STI
	CALL DELAY
	MOV BH, Power
	XOR BH, 01H
	MOV Power, BH
	MOV AL, 20H
	OUT 20H, AL ;中断结束命令
	IRET

DELAY: PUSH CX
	MOV CX, 0F00H
AA2: PUSH AX
	POP AX
	LOOP AA2
	POP CX
	RET

CODE ENDS
END START
