; 8255、DAC0832、8259波形发生器控制程序

STACK    SEGMENT STACK
    DW 256 DUP(?)
STACK    ENDS

DATA    SEGMENT
    STATUS      DB 0       ; 系统状态：0-关机，1-开机
    WAVE_TYPE   DB 0       ; 波形类型：0-方波，1-锯齿波，2-三角波
    FREQ_MODE   DB 0       ; 频率模式：0-正常，1-扩大一倍，2-缩小一倍
    DAC_PORT    DW 0600H   ; DAC0832端口地址
    PORT_A      EQU 60H    ; 8255端口A地址
    PORT_B      EQU 61H    ; 8255端口B地址
    PORT_C      EQU 62H    ; 8255端口C地址
    PORT_CTRL   EQU 63H    ; 8255控制端口地址
    WAVE_VALUE  DB 0       ; 当前波形值
    DIRECTION   DB 1       ; 三角波方向：1-递增，0-递减
DATA    ENDS

CODE    SEGMENT
    ASSUME CS:CODE, DS:DATA, SS:STACK

START:
    MOV AX, DATA
    MOV DS, AX
    
    ; 初始化8255
    MOV AL, 10010000B      ; 方式0，端口A输入
    OUT PORT_CTRL, AL
    
    ; 设置中断向量
    CLI
    PUSH DS
    MOV AX, 0000H
    MOV DS, AX
    MOV AX, OFFSET IRQ7_HANDLER
    MOV SI, 003CH
    MOV [SI], AX
    MOV AX, CS
    MOV SI, 003EH
    MOV [SI], AX
    POP DS
    
    ; 初始化8259
    MOV AL, 11H
    OUT 20H, AL
    MOV AL, 08H
    OUT 21H, AL
    MOV AL, 04H
    OUT 21H, AL
    MOV AL, 01H
    OUT 21H, AL
    MOV AL, 6FH
    OUT 21H, AL
    STI

MAIN_LOOP:
    ; 读取拨码开关状态
    IN AL, PORT_A
    AND AL, 07H
    
    ; 解析波形类型（低2位）
    MOV BL, AL
    AND BL, 03H
    MOV WAVE_TYPE, BL
    
    ; 解析频率模式（第3位）
    MOV BH, AL
    AND BH, 04H
    SHR BH, 2
    MOV FREQ_MODE, BH
    
    ; 检查系统状态
    CMP STATUS, 0
    JE SYSTEM_OFF
    
    ; 根据波形类型跳转
    CMP WAVE_TYPE, 0
    JE GENERATE_SQUARE
    CMP WAVE_TYPE, 1
    JE GENERATE_SAWTOOTH
    CMP WAVE_TYPE, 2
    JE GENERATE_TRIANGLE
    
    JMP MAIN_LOOP

SYSTEM_OFF:
    MOV DX, DAC_PORT
    MOV AL, 00H
    OUT DX, AL
    JMP MAIN_LOOP

GENERATE_SQUARE:
    MOV DX, DAC_PORT
    MOV AL, 0FFH
    OUT DX, AL
    CALL DELAY
    MOV AL, 00H
    OUT DX, AL
    CALL DELAY
    JMP MAIN_LOOP

GENERATE_SAWTOOTH:
    MOV CX, 0100H
    MOV DX, DAC_PORT
SAW_LOOP:
    MOV AL, CL
    DEC AL
    OUT DX, AL
    CALL DELAY
    LOOP SAW_LOOP
    JMP MAIN_LOOP

GENERATE_TRIANGLE:
    MOV DX, DAC_PORT
    MOV AL, WAVE_VALUE
    OUT DX, AL
    CMP DIRECTION, 1
    JE TRI_INC
    DEC AL
    CMP AL, 0
    JNE TRI_CONT
    MOV DIRECTION, 1
    JMP TRI_CONT
TRI_INC:
    INC AL
    CMP AL, 0FFH
    JNE TRI_CONT
    MOV DIRECTION, 0
TRI_CONT:
    MOV WAVE_VALUE, AL
    CALL DELAY
    JMP MAIN_LOOP

DELAY:
    PUSH CX
    PUSH AX
    MOV CX, 03FFH
DELAY_LOOP:
    PUSH AX
    POP AX
    LOOP DELAY_LOOP
    POP AX
    POP CX
    RET

IRQ7_HANDLER:
    PUSH AX
    PUSH DS
    MOV AX, DATA
    MOV DS, AX
    XOR STATUS, 01H
    MOV AL, 20H
    OUT 20H, AL
    POP DS
    POP AX
    IRET

CODE ENDS
END START
