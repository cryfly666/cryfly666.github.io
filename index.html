;==================================================
; 8255 + DAC0832 + 8259 波形发生器系统
; 编译环境：MASM 5.0 以上
; 硬件平台：8086/8088 实验箱
;==================================================

; 端口地址定义
PORTA       EQU 60H     ; 8255 A口地址
PORTB       EQU 61H     ; 8255 B口地址
PORTC       EQU 62H     ; 8255 C口地址
CTRL_8255   EQU 63H     ; 8255 控制口地址
DAC_PORT    EQU 64H     ; DAC0832 端口地址
ICW1_8259   EQU 20H     ; 8259 ICW1 地址
ICW2_8259   EQU 21H     ; 8259 ICW2/OCW1 地址
OCW2_8259   EQU 20H     ; 8259 OCW2 地址

; 数据段
DATA SEGMENT
    ; 正弦波数据表（32个点）
    SIN_TABLE DB 128, 150, 172, 192, 210, 226, 238, 246
              DB 250, 246, 238, 226, 210, 192, 172, 150
              DB 128, 106, 84, 64, 46, 30, 18, 10
              DB 6, 10, 18, 30, 46, 64, 84, 106
    
    ; 系统变量
    WAVE_TYPE DB 0       ; 波形类型: 0=方波, 1=三角波, 2=正弦波
    FREQUENCY DW 1000    ; 频率参数（延时计数）
    POWER_FLAG DB 1      ; 电源标志: 1=开, 0=关
    TEMP DB ?            ; 临时变量
DATA ENDS

; 堆栈段
STACK SEGMENT STACK
    DW 100 DUP(?)
STACK ENDS

; 代码段
CODE SEGMENT
    ASSUME CS:CODE, DS:DATA, SS:STACK

;==================================================
; 主程序开始
;==================================================
START:
    ; 初始化段寄存器
    MOV AX, DATA
    MOV DS, AX
    
    ; 初始化堆栈
    MOV AX, STACK
    MOV SS, AX
    MOV SP, 200
    
    ; 初始化8255
    CALL INIT_8255
    
    ; 初始化8259
    CALL INIT_8259
    
    ; 初始化DAC0832
    CALL INIT_DAC
    
    ; 设置中断向量
    CLI
    MOV AX, 0
    MOV ES, AX
    MOV DI, 20H * 4      ; 中断类型号20H（IR0）
    MOV AX, OFFSET POWER_ISR
    STOSW
    MOV AX, CS
    STOSW
    STI
    
    ; 主循环
MAIN_LOOP:
    ; 检查电源状态
    CMP POWER_FLAG, 0
    JE MAIN_LOOP         ; 如果关机，等待
    
    ; 读取按键
    CALL READ_KEY
    
    ; 根据波形类型输出
    CMP WAVE_TYPE, 0
    JE OUTPUT_SQUARE
    CMP WAVE_TYPE, 1
    JE OUTPUT_TRIANGLE
    CMP WAVE_TYPE, 2
    JE OUTPUT_SINE
    
    JMP MAIN_LOOP

;==================================================
; 输出方波
;==================================================
OUTPUT_SQUARE:
    ; 输出高电平
    MOV AL, 255
    OUT PORTA, AL
    CALL DELAY
    
    ; 输出低电平
    MOV AL, 0
    OUT PORTA, AL
    CALL DELAY
    
    JMP MAIN_LOOP

;==================================================
; 输出三角波
;==================================================
OUTPUT_TRIANGLE:
    ; 上升沿
    MOV AL, 0
TRI_UP:
    OUT PORTA, AL
    CALL SHORT_DELAY
    INC AL
    CMP AL, 255
    JNE TRI_UP
    
    ; 下降沿
    MOV AL, 255
TRI_DOWN:
    OUT PORTA, AL
    CALL SHORT_DELAY
    DEC AL
    CMP AL, 0
    JNE TRI_DOWN
    
    JMP MAIN_LOOP

;==================================================
; 输出正弦波
;==================================================
OUTPUT_SINE:
    MOV SI, OFFSET SIN_TABLE
    MOV CX, 32           ; 32个采样点
SINE_LOOP:
    MOV AL, [SI]
    OUT PORTA, AL
    CALL SHORT_DELAY
    INC SI
    LOOP SINE_LOOP
    
    JMP MAIN_LOOP

;==================================================
; 初始化8255
;==================================================
INIT_8255 PROC
    MOV AL, 10000010B    ; 方式0，A口输出，B口输入，C口输出
    OUT CTRL_8255, AL
    RET
INIT_8255 ENDP

;==================================================
; 初始化8259
;==================================================
INIT_8259 PROC
    ; ICW1
    MOV AL, 00010011B    ; 边沿触发，单片8259，需要ICW4
    OUT ICW1_8259, AL
    
    ; ICW2
    MOV AL, 20H          ; 中断类型号基址20H
    OUT ICW2_8259, AL
    
    ; ICW4
    MOV AL, 00000001B    ; 非缓冲，自动EOI，8086模式
    OUT ICW2_8259, AL
    
    ; OCW1 (屏蔽除IR0外的所有中断)
    MOV AL, 11111110B
    OUT ICW2_8259, AL
    
    RET
INIT_8259 ENDP

;==================================================
; 初始化DAC0832
;==================================================
INIT_DAC PROC
    ; 输出0V
    MOV AL, 0
    OUT DAC_PORT, AL
    RET
INIT_DAC ENDP

;==================================================
; 读取按键
;==================================================
READ_KEY PROC
    IN AL, PORTB         ; 读取B口状态
    
    ; 检查波形切换键 (PB0)
    TEST AL, 00000001B
    JNZ CHECK_FREQ_UP
    CALL CHANGE_WAVE
    
CHECK_FREQ_UP:
    ; 检查频率增加键 (PB1)
    TEST AL, 00000010B
    JNZ CHECK_FREQ_DOWN
    CALL INC_FREQ
    
CHECK_FREQ_DOWN:
    ; 检查频率减少键 (PB2)
    TEST AL, 00000100B
    JNZ KEY_DONE
    CALL DEC_FREQ
    
KEY_DONE:
    RET
READ_KEY ENDP

;==================================================
; 切换波形
;==================================================
CHANGE_WAVE PROC
    ; 去抖动延时
    CALL DELAY_10MS
    
    ; 切换波形类型
    INC WAVE_TYPE
    CMP WAVE_TYPE, 3
    JNE WAVE_OK
    MOV WAVE_TYPE, 0     ; 循环到0
    
WAVE_OK:
    ; 用LED指示波形类型
    MOV AL, WAVE_TYPE
    OUT PORTC, AL
    RET
CHANGE_WAVE ENDP

;==================================================
; 增加频率
;==================================================
INC_FREQ PROC
    CALL DELAY_10MS
    CMP FREQUENCY, 100
    JBE FREQ_MAX
    SUB FREQUENCY, 100   ; 减少延时，增加频率
FREQ_MAX:
    RET
INC_FREQ ENDP

;==================================================
; 减少频率
;==================================================
DEC_FREQ PROC
    CALL DELAY_10MS
    CMP FREQUENCY, 5000
    JAE FREQ_MIN
    ADD FREQUENCY, 100   ; 增加延时，减少频率
FREQ_MIN:
    RET
DEC_FREQ ENDP

;==================================================
; 中断服务程序（开关机控制）
;==================================================
POWER_ISR PROC FAR
    PUSH AX
    
    ; 切换电源状态
    NOT POWER_FLAG
    AND POWER_FLAG, 1
    
    ; 如果是关机，输出0V
    CMP POWER_FLAG, 0
    JNE POWER_ON
    MOV AL, 0
    OUT DAC_PORT, AL
    MOV AL, 0
    OUT PORTC, AL        ; 关闭LED
    
POWER_ON:
    ; 发送EOI命令
    MOV AL, 00100000B    ; 非特殊EOI
    OUT OCW2_8259, AL
    
    POP AX
    IRET
POWER_ISR ENDP

;==================================================
; 延时子程序
;==================================================
DELAY PROC
    PUSH CX
    MOV CX, FREQUENCY
DELAY_LOOP:
    LOOP DELAY_LOOP
    POP CX
    RET
DELAY ENDP

SHORT_DELAY PROC
    PUSH CX
    MOV CX, 10
SHORT_DELAY_LOOP:
    LOOP SHORT_DELAY_LOOP
    POP CX
    RET
SHORT_DELAY ENDP

DELAY_10MS PROC
    PUSH CX
    MOV CX, 1000
DELAY_10MS_LOOP:
    LOOP DELAY_10MS_LOOP
    POP CX
    RET
DELAY_10MS ENDP

;==================================================
; 程序结束
;==================================================
CODE ENDS
END START
