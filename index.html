; 端口地址定义
PORT8255_A  EQU 0280H   ; 8255 PA口（输出至DAC0832）
PORT8255_B  EQU 0281H   ; 8255 PB口（按键输入）
PORT8255_C  EQU 0282H   ; 8255 PC口
PORT8255_CTL EQU 0283H  ; 8255控制口
PORT_DAC0832 EQU 0290H  ; DAC0832数据口
PORT_8259_M EQU 20H     ; 主8259命令口
PORT_8259_IM EQU 21H    ; 主8259屏蔽口

; 中断向量表设置（IRQ2对应中断号0AH）
SET_VECTOR:
    CLI                 ; 关中断
    MOV AX,0
    MOV ES,AX
    MOV DI,0AH*4        ; 中断向量地址0000:0028H
    MOV AX,OFFSET INT8259 ; 中断服务程序偏移地址
    STOSW
    MOV AX,SEG INT8259  ; 中断服务程序段地址
    STOSW
    ; 8259初始化（边沿触发、单片、非自动EOI）
    MOV AL,13H
    OUT PORT_8259_M,AL
    MOV AL,08H
    OUT PORT_8259_IM,AL
    MOV AL,01H
    OUT PORT_8259_M,AL
    IN AL,PORT_8259_IM
    AND AL,0FBH         ; 开放IR2中断（bit2置0）
    OUT PORT_8259_IM,AL
    ; 8255初始化：PA输出，PB输入，方式0
    MOV AL,82H          ; 控制字：10000010B
    OUT PORT8255_CTL,AL
    STI                 ; 开中断
    RET

; 8259中断服务程序（开关机控制）
INT8259:
    PUSH AX
    IN AL,PORT_8259_IM
    OR AL,04H           ; 屏蔽IR2（避免重复触发）
    OUT PORT_8259_IM,AL
    ; 开关机逻辑：若系统运行则暂停，否则启动
    CMP FLAG_RUN,1
    JE  STOP_SYS
    MOV FLAG_RUN,1
    JMP INT_EXIT
STOP_SYS:
    MOV FLAG_RUN,0
INT_EXIT:
    MOV AL,20H          ; 发送EOI命令
    OUT PORT_8259_M,AL
    POP AX
    IRET                ; 中断返回
    ; 全局变量定义
FLAG_RUN   DB 1         ; 运行标志（1=运行，0=暂停）
WAVE_TYPE  DB 0         ; 波形类型（0=方波，1=三角波，2=正弦波）
FREQ_VAL   DB 5         ; 频率系数（越大频率越低）

; 主循环：读取按键+输出波形
MAIN_LOOP:
    CMP FLAG_RUN,0
    JE  MAIN_LOOP       ; 系统暂停则等待
    ; 读取8255 PB口按键
    IN AL,PORT8255_B
    AND AL,0FH          ; 只取PB0~PB3
    ; 波形切换（PB0/PB1）
    CMP AL,01H
    JE  SET_SQUARE      ; PB0按下=方波
    CMP AL,02H
    JE  SET_TRIANGLE    ; PB1按下=三角波
    CMP AL,04H
    JE  SET_SINE        ; PB2按下=正弦波
    ; 频率调节（PB2/PB3）
    CMP AL,08H
    JE  FREQ_UP         ; PB3按下=频率增加
    ; 根据波形类型输出数据
    CMP WAVE_TYPE,0
    JE  OUT_SQUARE
    CMP WAVE_TYPE,1
    JE  OUT_TRIANGLE
    CMP WAVE_TYPE,2
    JE  OUT_SINE
    JMP MAIN_LOOP

; 方波输出
OUT_SQUARE:
    MOV AL,0FFH         ; 高电平
    OUT PORT8255_A,AL
    OUT PORT_DAC0832,AL
    CALL DELAY          ; 延时
    MOV AL,00H          ; 低电平
    OUT PORT8255_A,AL
    OUT PORT_DAC0832,AL
    CALL DELAY
    JMP MAIN_LOOP

; 三角波输出（简化版）
OUT_TRIANGLE:
    MOV CX,0FFH
TRI_UP:
    MOV AL,CL
    OUT PORT8255_A,AL
    OUT PORT_DAC0832,AL
    CALL DELAY
    LOOP TRI_UP
    MOV CX,0FFH
TRI_DOWN:
    MOV AL,CL
    OUT PORT8255_A,AL
    OUT PORT_DAC0832,AL
    CALL DELAY
    LOOP TRI_DOWN
    JMP MAIN_LOOP

; 正弦波输出（预存量化表，取32个点）
SINE_TABLE DB 80H,96H,0ADH,0C3H,0D8H,0EAH,0F9H,0FFH
           DB 0FFH,0F9H,0EAH,0D8H,0C3H,0ADH,96H,80H
           DB 67H,50H,3AH,25H,13H,04H,00H,00H
           DB 04H,13H,25H,3AH,50H,67H,80H,00H
OUT_SINE:
    LEA SI,SINE_TABLE
    MOV CX,32
SINE_LOOP:
    MOV AL,[SI]
    OUT PORT8255_A,AL
    OUT PORT_DAC0832,AL
    CALL DELAY
    INC SI
    LOOP SINE_LOOP
    JMP MAIN_LOOP

; 延时函数（由频率系数控制）
DELAY:
    PUSH CX
    MOV CL,FREQ_VAL
DELAY1:
    PUSH CX
    MOV CX,0FFFFH
DELAY2:
    LOOP DELAY2
    POP CX
    LOOP DELAY1
    POP CX
    RET

; 按键处理子程序（波形切换/频率调节）
SET_SQUARE:
    MOV WAVE_TYPE,0
    RET
SET_TRIANGLE:
    MOV WAVE_TYPE,1
    RET
SET_SINE:
    MOV WAVE_TYPE,2
    RET
FREQ_UP:
    CMP FREQ_VAL,1
    JE  FREQ_EXIT
    DEC FREQ_VAL        ; 频率系数减小，频率升高
FREQ_EXIT:
    RET
